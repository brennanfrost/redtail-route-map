    point: { pixelSize: 9, color: C.Color.WHITE, outlineColor: C.Color.fromCssColorString('#3a2a22'), outlineWidth: 2, disableDepthTestDistance: Number.POSITIVE_INFINITY },
    label: {
      text: '✈ ' + home.name, font: '600 13px ' + FONT, fillColor: C.Color.WHITE, showBackground: true,
      backgroundColor: DARK, backgroundPadding: new C.Cartesian2(9, 5),
      verticalOrigin: C.VerticalOrigin.BOTTOM, pixelOffset: new C.Cartesian2(0, -12),
      disableDepthTestDistance: Number.POSITIVE_INFINITY
    }
  });

  /* ---------- camera ---------- */
  const center = { lat: 0, lon: 0 };
  pts.forEach(p => { center.lat += p.lat / pts.length; center.lon += p.lon / pts.length; });
  let spread = 0; pts.forEach(p => spread = Math.max(spread, km(center, p)));
  const OVERVIEW = { yaw: -15, pitch: -48, range: spread * 1000 * 2.3 };

  const user = { yaw: 0, pitch: CONFIG.orbitPitchDeg, zoom: 1, ovYaw: 0, ovPitch: OVERVIEW.pitch, ovZoom: 1 };
  let phase = 'overview';  // 'overview' | 'fly' | 'orbit'
  let orbitStart = 0, orbitAngle = 0, flight = null;

  const overviewPose = () => ({
    target: C.Cartesian3.fromDegrees(center.lon, center.lat, DEFAULT_H),
    heading: OVERVIEW.yaw + user.ovYaw, pitch: user.ovPitch, range: OVERVIEW.range * user.ovZoom
  });
  const orbitPose = i => ({
    target: STOPS[i].pos,
    heading: orbitStart + orbitAngle + user.yaw, pitch: user.pitch, range: STOPS[i].range * user.zoom
  });

  function pose(now) {
    if (phase === 'toOverview') return lastPose;
    if (phase === 'overview') return overviewPose();
    if (phase === 'orbit') return orbitPose(current);
    const u = clamp((now - flight.t0) / flight.dur, 0, 1), e = ease(u);
    const to = orbitPose(current), f = flight.from;
    const lr = Math.log(f.range) + (Math.log(to.range) - Math.log(f.range)) * e + Math.sin(Math.PI * u) * flight.hop;
    if (u >= 1) { phase = 'orbit'; }
    return {
      target: C.Cartesian3.lerp(f.target, to.target, e, new C.Cartesian3()),
      heading: f.heading + angDiff(f.heading, to.heading) * e,
      pitch: f.pitch + (to.pitch - f.pitch) * e,
      range: Math.exp(lr)
    };
  }
  let lastPose = overviewPose();
  scene.preRender.addEventListener(() => {
    const p = pose(performance.now());
    lastPose = p;
    camera.lookAtTransform(C.Transforms.eastNorthUpToFixedFrame(p.target),
      new C.HeadingPitchRange(rad(p.heading), rad(p.pitch), p.range));
  });

  const reduceMotion = () => matchMedia('(prefers-reduced-motion: reduce)').matches;
  function flyTo(i) {
    const from = lastPose;
    const prev = current >= 0 ? STOPS[current] : center;
    current = i;
    orbitStart = bearing(prev, STOPS[i]);
    if (current >= 0 && prev === STOPS[i]) orbitStart = from.heading;
    orbitAngle = 0; user.yaw = 0;
    const dist = km(prev, STOPS[i]);
    const fromOverview = from.range > STOPS[i].range * 2.5;
    const dur = reduceMotion() ? 1 : clamp(4200 + dist * 380, 4500, 12000) / Number($('speed').value || 1);
    const hop = fromOverview ? 0 : clamp(Math.log((dist * 1000) / (STOPS[i].range * 3)), 0, 0.45);
    flight = { from, t0: performance.now(), dur, hop };
    phase = 'fly';
    updateUI();
  }
  function toOverview() {
    const from = lastPose;
    current = -1;
    const t0 = performance.now(), dur = reduceMotion() ? 1 : 3200;
    phase = 'toOverview';
    const step = () => {
      if (phase !== 'toOverview') return;
      const u = clamp((performance.now() - t0) / dur, 0, 1), e = ease(u);
      const tgt = overviewPose();
      lastPose = {
        target: C.Cartesian3.lerp(from.target, tgt.target, e, new C.Cartesian3()),
        heading: from.heading + angDiff(from.heading, tgt.heading) * e,
        pitch: from.pitch + (tgt.pitch - from.pitch) * e,
        range: Math.exp(Math.log(from.range) + (Math.log(tgt.range) - Math.log(from.range)) * e)
      };
      if (u >= 1) { phase = 'overview'; return; }
      requestAnimationFrame(step);
    };
    requestAnimationFrame(step);
    updateUI();
  }

  /* ---------- touch / mouse: one finger turns and tilts, two fingers zoom ---------- */
  const area = $('map');
  const hideHint = () => $('hint').classList.add('gone');
  setTimeout(hideHint, 6000);
  const ov = () => phase === 'overview' || phase === 'toOverview';
  const SIGN = CONFIG.invertDrag ? -1 : 1;
  function turn(dx, dy) {
    if (ov()) { user.ovYaw -= SIGN * dx * 0.3; user.ovPitch = clamp(user.ovPitch - SIGN * dy * 0.25, -89, -5); }
    else { user.yaw -= SIGN * dx * 0.3; user.pitch = clamp(user.pitch - SIGN * dy * 0.25, -89, -3); }
  }
  function zoomBy(f) {
    if (ov()) user.ovZoom = clamp(user.ovZoom * f, 0.25, 2.5);
    else user.zoom = clamp(user.zoom * f, 0.2, 4);
  }

  // Touch (phones/tablets)
  let t1 = null, pinch0 = 0;
  const tdist = tl => Math.hypot(tl[0].clientX - tl[1].clientX, tl[0].clientY - tl[1].clientY);
  const onControl = e => e.target && e.target.closest && e.target.closest('button, a');
  function tStart(e) {
    if (onControl(e)) return;
    hideHint();
    if (e.touches.length === 1) { t1 = { x: e.touches[0].clientX, y: e.touches[0].clientY }; pinch0 = 0; }
    else if (e.touches.length >= 2) { t1 = null; pinch0 = tdist(e.touches); }
    e.preventDefault();
  }
  function tMove(e) {
    if (onControl(e)) return;
    e.preventDefault();
    if (e.touches.length === 1 && t1) {
      const x = e.touches[0].clientX, y = e.touches[0].clientY;
      turn(x - t1.x, y - t1.y);
      t1 = { x, y };
    } else if (e.touches.length >= 2) {
      const d = tdist(e.touches);
      if (pinch0 > 0 && d > 0) zoomBy(pinch0 / d);
      pinch0 = d;
    }
  }
  function tEnd(e) {
    if (e.touches.length === 1) { t1 = { x: e.touches[0].clientX, y: e.touches[0].clientY }; pinch0 = 0; }
    else if (e.touches.length === 0) { t1 = null; pinch0 = 0; }
  }
  area.addEventListener('touchstart', tStart, { passive: false, capture: true });
  area.addEventListener('touchmove', tMove, { passive: false, capture: true });
  area.addEventListener('touchend', tEnd, { capture: true });
  area.addEventListener('touchcancel', tEnd, { capture: true });

  // Mouse (computers)
  let m1 = null;
  area.addEventListener('pointerdown', e => {
    if (e.pointerType !== 'mouse' || onControl(e)) return;
    hideHint(); m1 = { x: e.clientX, y: e.clientY };
    try { area.setPointerCapture(e.pointerId); } catch (_) {}
  }, true);
  area.addEventListener('pointermove', e => {
    if (e.pointerType !== 'mouse' || !m1) return;
    turn(e.clientX - m1.x, e.clientY - m1.y);
    m1 = { x: e.clientX, y: e.clientY };
  }, true);
  const mUp = e => { if (e.pointerType === 'mouse') m1 = null; };
  area.addEventListener('pointerup', mUp, true);
  area.addEventListener('pointercancel', mUp, true);
  area.addEventListener('contextmenu', e => e.preventDefault());
  let toastTimer;
  area.addEventListener('wheel', e => {
    if (!(e.ctrlKey || e.metaKey)) {
      $('toast').textContent = 'Hold Ctrl or ⌘ and scroll to zoom';
      $('toast').classList.add('on');
      clearTimeout(toastTimer);
      toastTimer = setTimeout(() => $('toast').classList.remove('on'), 1400);
      return;
    }
    e.preventDefault(); hideHint();
    zoomBy(Math.exp(e.deltaY * 0.004));
  }, { passive: false });

  /* ---------- tour playback ---------- */
  let playing = false, last = null;
  function setPlaying(on) {
    playing = on;
    root.classList.toggle('playing', on);
    root.classList.toggle('min', on);
    $('playIcon').innerHTML = on ? ICON_PAUSE : ICON_PLAY;
    $('play').setAttribute('aria-label', on ? 'Pause the tour' : (current >= 0 ? 'Resume the tour' : 'Start the tour'));
    const mp = $('mpause');
    mp.querySelector('svg').innerHTML = on ? ICON_PAUSE : ICON_PLAY;
    mp.setAttribute('aria-label', on ? 'Pause' : 'Play');
    if (on) { last = null; requestAnimationFrame(tick); }
  }
  function tick(ts) {
    if (!playing) return;
    if (last !== null && phase === 'orbit') {
      const dt = Math.min((ts - last) / 1000, 0.1);
      orbitAngle += dt * 360 / CONFIG.orbitSeconds * Number($('speed').value || 1);
      ring();
      if (orbitAngle >= 360) {
        if (current < N - 1) flyTo(current + 1);
        else { setPlaying(false); toOverview(); }
      }
    }
    last = ts;
    if (playing) requestAnimationFrame(tick);
  }
  function ring() {
    const f = phase === 'orbit' ? clamp(orbitAngle / 360, 0, 1) : 0;
    $('ring').setAttribute('stroke-dashoffset', String(56.55 * (1 - f)));
  }

  function updateUI() {
    const on = current >= 0;
    $('now').classList.toggle('off', !on);
    if (on) {
      $('nowNum').textContent = 'Stop ' + (current + 1) + ' of ' + N;
      $('nowName').textContent = STOPS[current].name;
      $('mnum').textContent = 'Stop ' + (current + 1) + ' of ' + N;
      $('mname').textContent = STOPS[current].name;
      $('subtitle').textContent = 'Stop ' + (current + 1) + ' of ' + N + ': ' + STOPS[current].name;
    } else {
      $('subtitle').textContent = CONFIG.subtitle;
      $('mnum').textContent = 'Overview'; $('mname').textContent = CONFIG.title;
    }
    chips.forEach((c, i) => c.classList.toggle('on', i === current));
    if (on) chips[current].scrollIntoView({ behavior: reduceMotion() ? 'auto' : 'smooth', inline: 'center', block: 'nearest' });
    ring();
  }

  function play() {
    if (current < 0 || phase === 'overview' || phase === 'toOverview') flyTo(current < 0 ? 0 : current);
    setPlaying(true);
  }
  function go(i) { flyTo(clamp(i, 0, N - 1)); }

  $('play').addEventListener('click', () => playing ? setPlaying(false) : play());
  $('mpause').addEventListener('click', () => playing ? setPlaying(false) : play());
  $('expand').addEventListener('click', () => root.classList.remove('min'));
  $('collapse').addEventListener('click', () => root.classList.add('min'));
  $('next').addEventListener('click', () => go(current + 1));
  $('mnext').addEventListener('click', () => go(current + 1));
  $('prev').addEventListener('click', () => go(current - 1));
  $('overview').addEventListener('click', () => { setPlaying(false); toOverview(); });
  chips.forEach((c, i) => c.addEventListener('click', () => go(i)));

  updateUI();
}

loadLib(0);
})();
</script>
