const rad = d => d * Math.PI / 180, deg = r => r * 180 / Math.PI;
const clamp = (v, a, b) => Math.min(b, Math.max(a, v));
const angDiff = (a, b) => ((b - a + 540) % 360) - 180;
const ease = u => 0.5 - 0.5 * Math.cos(Math.PI * u);
const esc = t => String(t).replace(/[&<>"]/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]));
function bearing(a, b) {
  const y = Math.sin(rad(b.lon-a.lon)) * Math.cos(rad(b.lat));
  const x = Math.cos(rad(a.lat))*Math.sin(rad(b.lat)) - Math.sin(rad(a.lat))*Math.cos(rad(b.lat))*Math.cos(rad(b.lon-a.lon));
  return (deg(Math.atan2(y, x)) + 360) % 360;
}
function km(a, b) {
  const dLat = rad(b.lat - a.lat), dLon = rad(b.lon - a.lon);
  const h = Math.sin(dLat/2)**2 + Math.cos(rad(a.lat))*Math.cos(rad(b.lat))*Math.sin(dLon/2)**2;
  return 12742 * Math.asin(Math.sqrt(h));
}

function loadLib(i) {
  if (window.Cesium) return start();
  if (i >= LIB_SOURCES.length) return fail('The 3D map couldn’t load. Check your connection and try again.');
  window.CESIUM_BASE_URL = LIB_SOURCES[i];
  const css = document.createElement('link');
  css.rel = 'stylesheet'; css.href = LIB_SOURCES[i] + 'Widgets/widgets.css';
  document.head.appendChild(css);
  const js = document.createElement('script');
  js.src = LIB_SOURCES[i] + 'Cesium.js';
  js.onload = start;
  js.onerror = () => { css.remove(); js.remove(); loadLib(i + 1); };
  document.head.appendChild(js);
}

async function start() {
  if (!hasWebGL()) return fail('This map needs WebGL, which is off or unavailable in this browser. Try a current version of Chrome, Safari, Edge or Firefox.');
  if (!CONFIG.googleApiKey || CONFIG.googleApiKey.indexOf('PASTE_') === 0)
    return fail('Add your Google Maps API key to the <b>googleApiKey</b> setting to show the 3D imagery.');

  const C = window.Cesium;
  C.GoogleMaps.defaultApiKey = CONFIG.googleApiKey;

  let viewer;
  try {
    viewer = new C.Viewer($('map'), {
      globe: false, baseLayer: false, baseLayerPicker: false, geocoder: false, homeButton: false,
      sceneModePicker: false, navigationHelpButton: false, animation: false, timeline: false,
      fullscreenButton: false, infoBox: false, selectionIndicator: false, requestRenderMode: false
    });
  } catch (e) {
    console.error(e);
    return fail('The 3D map couldn’t start: ' + (e && e.message ? e.message : 'unknown error'));
  }
  const scene = viewer.scene, camera = viewer.camera, canvas = viewer.canvas;
  scene.skyAtmosphere.show = true;
  scene.screenSpaceCameraController.enableInputs = false;   // all camera control is handled below
  // sharper text and imagery on high-density screens (capped for smooth performance)
  viewer.useBrowserRecommendedResolution = false;
  viewer.resolutionScale = Math.min(1, 1.5 / (window.devicePixelRatio || 1));
  viewer.screenSpaceEventHandler.removeInputAction(C.ScreenSpaceEventType.LEFT_DOUBLE_CLICK);
  viewer.screenSpaceEventHandler.removeInputAction(C.ScreenSpaceEventType.LEFT_CLICK);

  /* Credits: Google logo on screen; data sources behind a small "Data" toggle */
  const dataBtn = document.createElement('button');
  dataBtn.className = 'rt-data'; dataBtn.type = 'button'; dataBtn.textContent = 'Data';
  dataBtn.setAttribute('aria-expanded', 'false');
  dataBtn.addEventListener('click', () => dataBtn.setAttribute('aria-expanded', String(root.classList.toggle('show-credits'))));
  viewer.cesiumWidget.creditContainer.appendChild(dataBtn);

  try {
    const tiles = await C.createGooglePhotorealistic3DTileset(undefined, { showCreditsOnScreen: true });
    tiles.maximumScreenSpaceError = 20;
    tiles.cacheBytes = 768 * 1024 * 1024;
    scene.primitives.add(tiles);
    let warned = false;
    tiles.tileFailed.addEventListener(err => {
      if (warned || !/40[13]|key|denied|forbidden/i.test(String(err && (err.message || err)))) return;
      warned = true; fail(KEY_MSG);
    });
  } catch (e) {
    console.error(e);
    return fail(KEY_MSG);
  }

  /* ---------- landmark positions (ground height measured from the 3D imagery, eased in) ---------- */
  const home = CONFIG.home;
  const pts = STOPS.concat([home]);
  pts.forEach(p => { p.h = p.hT = DEFAULT_H; p.pos = C.Cartesian3.fromDegrees(p.lon, p.lat, p.h); });
  scene.preUpdate.addEventListener(() => pts.forEach(p => {
    if (Math.abs(p.hT - p.h) < 0.5) return;
    p.h += (p.hT - p.h) * 0.05;
    p.pos = C.Cartesian3.fromDegrees(p.lon, p.lat, p.h);
  }));
  if (scene.sampleHeightMostDetailed) {
    scene.sampleHeightMostDetailed(pts.map(p => C.Cartographic.fromDegrees(p.lon, p.lat)))
      .then(res => res.forEach((c, i) => { if (c && isFinite(c.height)) pts[i].hT = c.height; }))
      .catch(e => console.warn('Height sampling failed', e));
  }

  /* ---------- labels (tap to learn more) ---------- */
  let current = -1, selected = 0;
  const RED = C.Color.fromCssColorString('#d42a2f');
  const DARK = C.Color.fromCssColorString('rgba(24,19,16,0.8)');
  const FONT = '"Barlow Semi Condensed", sans-serif';
  const PosProp = C.CallbackPositionProperty || C.CallbackProperty;
  const entityStop = new Map();
  STOPS.forEach((s, i) => {
    const ent = viewer.entities.add({
      position: new PosProp(() => s.pos, false),
      point: {
        pixelSize: new C.CallbackProperty(() => i === current ? 12 : 9, false),
        color: new C.CallbackProperty(() => i === current ? RED : C.Color.WHITE, false),
        outlineColor: new C.CallbackProperty(() => i === current ? C.Color.WHITE : RED, false),
        outlineWidth: 2, disableDepthTestDistance: Number.POSITIVE_INFINITY
      },
      label: {
        text: s.name,
        font: '700 30px ' + FONT, scale: new C.CallbackProperty(() => i === current ? 0.6 : 0.5, false),
        fillColor: C.Color.WHITE, style: C.LabelStyle.FILL, showBackground: true,
        backgroundColor: new C.CallbackProperty(() => i === current ? RED : DARK, false),
        backgroundPadding: new C.Cartesian2(20, 12),
        verticalOrigin: C.VerticalOrigin.BOTTOM, pixelOffset: new C.Cartesian2(0, -12),
        disableDepthTestDistance: Number.POSITIVE_INFINITY
      }
    });
    entityStop.set(ent, i);
  });
  viewer.entities.add({
    position: new PosProp(() => home.pos, false),
    point: { pixelSize: 9, color: C.Color.WHITE, outlineColor: C.Color.fromCssColorString('#3a2a22'), outlineWidth: 2, disableDepthTestDistance: Number.POSITIVE_INFINITY },
    label: {
      text: '✈ ' + home.name, font: '600 26px ' + FONT, scale: 0.5, fillColor: C.Color.WHITE, style: C.LabelStyle.FILL, showBackground: true,
      backgroundColor: DARK, backgroundPadding: new C.Cartesian2(18, 10),
      verticalOrigin: C.VerticalOrigin.BOTTOM, pixelOffset: new C.Cartesian2(0, -12),
      disableDepthTestDistance: Number.POSITIVE_INFINITY
    }
  });
  // Tap/click hit test: checks each landmark's label box on screen (more reliable than 3D picking)
  const toScreen = C.SceneTransforms.worldToWindowCoordinates || C.SceneTransforms.wgs84ToWindowCoordinates;
  function pickStop(x, y) {
    const r = canvas.getBoundingClientRect();
    const px = x - r.left, py = y - r.top;
    let best = -1, bestD = Infinity;
    STOPS.forEach((s, i) => {
      const p = toScreen(scene, s.pos);
      if (!p) return;
      const sc = i === current ? 0.6 : 0.5;
      const halfW = (s.name.length * 15 + 40) * sc / 2 + 6;
      const top = p.y - 12 - (30 + 24) * sc - 6, bottom = p.y + 10;
      if (px >= p.x - halfW && px <= p.x + halfW && py >= top && py <= bottom) {
        const d = Math.abs(px - p.x) + Math.abs(py - (top + bottom) / 2);
        if (d < bestD) { bestD = d; best = i; }
      }
    });
    if (best >= 0) { showInfo(best); return true; }
    return false;
  }

  /* ---------- camera ---------- */
  // phases: 'explore' (free map) | 'fly' (to a landmark) | 'orbit' (circling) | 'toTop' (back to overhead)
  let phase = 'explore', orbitStart = 0, orbitAngle = 0, anim = null, lastPose = null, free = null;
  const user = { yaw: 0, pitch: CONFIG.orbitPitchDeg, zoom: 1 };

  function topPose() {
    let w = 180, e = -180, s = 90, n = -90;
    pts.forEach(p => { w = Math.min(w, p.lon); e = Math.max(e, p.lon); s = Math.min(s, p.lat); n = Math.max(n, p.lat); });
    const px = (e - w) * 0.12, py = (n - s) * 0.08;
    const rect = C.Rectangle.fromDegrees(w - px, s - py * 3, e + px, n + py);
    const dest = camera.getRectangleCameraCoordinates(rect);
    const c = C.Cartographic.fromCartesian(dest);
    return {
      target: C.Cartesian3.fromDegrees(C.Math.toDegrees(c.longitude), C.Math.toDegrees(c.latitude), DEFAULT_H),
      heading: 0, pitch: -89.5, range: clamp(c.height - DEFAULT_H, 5000, 400000)
    };
  }
  const TOP = topPose();
  const orbitPose = i => ({
    target: STOPS[i].pos,
    heading: orbitStart + orbitAngle + user.yaw, pitch: user.pitch, range: STOPS[i].range * user.zoom
  });
  function blend(f, to, u, hop) {
    const e = ease(u);
    const lr = Math.log(f.range) + (Math.log(to.range) - Math.log(f.range)) * e + Math.sin(Math.PI * u) * (hop || 0);
    return {
      target: C.Cartesian3.lerp(f.target, to.target, e, new C.Cartesian3()),
      heading: f.heading + angDiff(f.heading, to.heading) * e,
      pitch: f.pitch + (to.pitch - f.pitch) * e,
      range: Math.exp(lr)
    };
  }
  function applyPose(p) {
    camera.lookAtTransform(C.Transforms.eastNorthUpToFixedFrame(p.target),
      new C.HeadingPitchRange(rad(p.heading), rad(p.pitch), p.range));
    lastPose = p;
  }
  function enterExplore(p) {
    free = { target: p.target.clone ? p.target.clone() : p.target, heading: p.heading, pitch: p.pitch, range: p.range };
    phase = 'explore';
  }

  scene.preRender.addEventListener(() => {
    if (phase === 'explore') return applyPose(free);
    if (phase === 'orbit') return applyPose(orbitPose(current));
    const u = clamp((performance.now() - anim.t0) / anim.dur, 0, 1);
    if (phase === 'fly') {
      applyPose(blend(anim.from, orbitPose(current), u, anim.hop));
      if (u >= 1) phase = 'orbit';
    } else if (phase === 'toTop') {
      const p = blend(anim.from, TOP, u, 0);
      applyPose(p);
      if (u >= 1) enterExplore(TOP);
    }
  });

  const reduceMotion = () => matchMedia('(prefers-reduced-motion: reduce)').matches;

  function flyTo(i) {
    i = clamp(i, 0, N - 1);
    const from = lastPose;
    const prev = current >= 0 ? STOPS[current] : null;
    current = selected = i;
    orbitAngle = 0; user.yaw = 0;
    if (prev && prev !== STOPS[i]) orbitStart = bearing(prev, STOPS[i]);
    else orbitStart = from.heading;
    const dist = C.Cartesian3.distance(from.target, STOPS[i].pos) / 1000;
    const fromHigh = from.range > STOPS[i].range * 2.5;
    const dur = reduceMotion() ? 1 : clamp(4200 + dist * 380, 4500, 12000);
    const hop = fromHigh ? 0 : clamp(Math.log((dist * 1000) / (STOPS[i].range * 3)), 0, 0.45);
    anim = { from, t0: performance.now(), dur, hop };
    phase = 'fly';
    updateUI();
  }
  function toTop() {
    current = -1;
    anim = { from: lastPose, t0: performance.now(), dur: reduceMotion() ? 1 : 3400 };
    phase = 'toTop';
    updateUI();
  }

  /* ---------- touch & mouse ----------
     Exploring: one finger moves the map, pinch zooms, two fingers slide to tilt and twist to turn.
     Circling a landmark: one finger turns and tilts the view, pinch zooms. Tap a name for its info. */
  const area = $('map');
  const SIGN = CONFIG.invertDrag ? -1 : 1;
  const onControl = e => e.target && e.target.closest && e.target.closest('button, a');
  const settle = () => {
    if (phase === 'toTop') enterExplore(lastPose);
  };
  function pan(dx, dy) {
    if (phase !== 'explore') return;
    const mpp = 2 * free.range * Math.tan((camera.frustum.fovy || 1) / 2) / Math.max(canvas.clientHeight, 1);
    const h = rad(free.heading), tilt = Math.max(Math.sin(-rad(free.pitch)), 0.35);
    const fwd = SIGN * dy * mpp / tilt, right = -SIGN * dx * mpp;
    const e = right * Math.cos(h) + fwd * Math.sin(h);
    const n = -right * Math.sin(h) + fwd * Math.cos(h);
    free.target = C.Matrix4.multiplyByPoint(C.Transforms.eastNorthUpToFixedFrame(free.target), new C.Cartesian3(e, n, 0), new C.Cartesian3());
  }
  function turn(dx, dy) {
    if (phase === 'explore') {
      free.heading -= SIGN * dx * 0.3;
      free.pitch = clamp(free.pitch - SIGN * dy * 0.25, -89.5, -8);
    } else {
      user.yaw -= SIGN * dx * 0.3;
      user.pitch = clamp(user.pitch - SIGN * dy * 0.25, -89, -3);
    }
  }
  function tilt(dy) {
    if (phase === 'explore') free.pitch = clamp(free.pitch + SIGN * dy * 0.25, -89.5, -8);
    else user.pitch = clamp(user.pitch + SIGN * dy * 0.25, -89, -3);
  }
  function twist(da) {
    if (phase === 'explore') free.heading -= da; else user.yaw -= da;
  }
  function zoomBy(f) {
    if (phase === 'explore') free.range = clamp(free.range * f, 250, TOP.range * 1.6);
    else user.zoom = clamp(user.zoom * f, 0.04, 4);
  }

  // Touch
  let one = null, two = null, tap = null;
  const twoState = tl => {
    const a = tl[0], b = tl[1];
    return { d: Math.hypot(a.clientX - b.clientX, a.clientY - b.clientY), ang: deg(Math.atan2(b.clientY - a.clientY, b.clientX - a.clientX)), my: (a.clientY + b.clientY) / 2 };
  };
  area.addEventListener('touchstart', e => {
    if (onControl(e)) return;
    e.preventDefault();
    settle();
    if (e.touches.length === 1) {
      one = { x: e.touches[0].clientX, y: e.touches[0].clientY }; two = null;
      tap = { x: one.x, y: one.y, t: performance.now() };
    } else if (e.touches.length >= 2) { one = null; tap = null; two = twoState(e.touches); }
  }, { passive: false, capture: true });
  area.addEventListener('touchmove', e => {
    if (onControl(e)) return;
    e.preventDefault();
    if (e.touches.length === 1 && one) {
      const x = e.touches[0].clientX, y = e.touches[0].clientY;
      if (phase === 'explore') pan(x - one.x, y - one.y); else turn(x - one.x, y - one.y);
      one = { x, y };
      if (tap && Math.hypot(x - tap.x, y - tap.y) > 8) tap = null;
    } else if (e.touches.length >= 2 && two) {
      const n = twoState(e.touches);
      if (n.d > 0 && two.d > 0) zoomBy(two.d / n.d);
      tilt(n.my - two.my);
      twist(angDiff(two.ang, n.ang));
      two = n;
    }
  }, { passive: false, capture: true });
  const tEnd = e => {
    if (e.touches.length === 0) {
      if (tap && performance.now() - tap.t < 400) pickStop(tap.x, tap.y);
      one = two = tap = null;
    } else if (e.touches.length === 1) {
      one = { x: e.touches[0].clientX, y: e.touches[0].clientY }; two = null; tap = null;
    }
  };
  area.addEventListener('touchend', tEnd, { capture: true });
  area.addEventListener('touchcancel', tEnd, { capture: true });

  // Mouse
  let m = null;
  area.addEventListener('pointerdown', e => {
    if (e.pointerType !== 'mouse' || onControl(e)) return;
    settle();
    m = { x: e.clientX, y: e.clientY, sx: e.clientX, sy: e.clientY, t: performance.now(), rot: e.button === 2 || e.ctrlKey || e.metaKey };
    try { area.setPointerCapture(e.pointerId); } catch (_) {}
  }, true);
  area.addEventListener('pointermove', e => {
    if (e.pointerType !== 'mouse' || !m) return;
    const dx = e.clientX - m.x, dy = e.clientY - m.y;
    if (phase === 'explore' && !m.rot) pan(dx, dy); else turn(dx, dy);
    m.x = e.clientX; m.y = e.clientY;
  }, true);
  const mUp = e => {
    if (e.pointerType !== 'mouse' || !m) return;
    if (Math.hypot(e.clientX - m.sx, e.clientY - m.sy) < 5 && performance.now() - m.t < 500) pickStop(e.clientX, e.clientY);
    m = null;
  };
  area.addEventListener('pointerup', mUp, true);
  area.addEventListener('pointercancel', () => { m = null; }, true);
  area.addEventListener('contextmenu', e => e.preventDefault());

  let toastTimer;
  area.addEventListener('wheel', e => {
    if (!(e.ctrlKey || e.metaKey)) {
      $('toast').textContent = 'Hold Ctrl or ⌘ and scroll to zoom';
      $('toast').classList.add('on');
      clearTimeout(toastTimer);
      toastTimer = setTimeout(() => $('toast').classList.remove('on'), 1400);
      return;
    }
    e.preventDefault();
    settle();
    zoomBy(Math.exp(e.deltaY * 0.004));
  }, { passive: false, capture: true });

  /* ---------- info sheet ---------- */
  let infoIdx = -1;
  function showInfo(i) {
    const s = STOPS[i];
    infoIdx = i; selected = i;
    $('sheetNum').textContent = 'Stop ' + (i + 1) + ' of ' + N;
    $('sheetTitle').textContent = s.name;
    $('sheetBody').innerHTML = '<p>' + esc(s.about) + '</p><ul>' + s.facts.map(f => '<li>' + esc(f) + '</li>').join('') + '</ul>';
    $('sheetBody').scrollTop = 0;
    $('sheetFly').textContent = playing ? 'Continue tour from here' : 'Fly here';
    $('sheetFly').hidden = (i === current && phase !== 'explore');
    root.classList.add('info');
    updateUI();
  }
  const hideInfo = () => { root.classList.remove('info'); infoIdx = -1; };
  $('sheetClose').addEventListener('click', hideInfo);
  $('sheetFly').addEventListener('click', () => { const i = infoIdx; hideInfo(); flyTo(i); });

  /* ---------- tour playback ---------- */
  let playing = false, last = null;
  function setPlaying(on) {
    playing = on;
    root.classList.toggle('playing', on);
    root.classList.toggle('min', on);
    $('playIcon').innerHTML = on ? ICON_PAUSE : ICON_PLAY;
    $('play').setAttribute('aria-label', on ? 'Pause the tour' : 'Play the tour');
    const mp = $('mpause');
    mp.querySelector('svg').innerHTML = on ? ICON_PAUSE : ICON_PLAY;
    mp.setAttribute('aria-label', on ? 'Pause' : 'Play');
    if (on) { last = null; requestAnimationFrame(tick); }
  }
  function tick(ts) {
    if (!playing) return;
    if (last !== null && phase === 'orbit') {
      const dt = Math.min((ts - last) / 1000, 0.1);
      orbitAngle += dt * 360 / CONFIG.orbitSeconds;
      ring();
      if (orbitAngle >= 360) {
        if (current < N - 1) flyTo(current + 1);
        else { setPlaying(false); toTop(); }
      }
    }
    last = ts;
    if (playing) requestAnimationFrame(tick);
  }
  function ring() {
    const f = phase === 'orbit' ? clamp(orbitAngle / 360, 0, 1) : 0;
    $('ring').setAttribute('stroke-dashoffset', String(56.55 * (1 - f)));
  }
  function updateUI() {
    const i = current >= 0 ? current : selected;
    if (current >= 0) {
      $('mnum').textContent = 'Stop ' + (current + 1) + ' of ' + N;
      $('mname').textContent = STOPS[current].name;
    } else {
      $('mnum').textContent = 'Overview';
      $('mname').textContent = CONFIG.title;
    }
    chips.forEach((c, k) => c.classList.toggle('on', k === current));
    centerChip(i);
    ring();
  }

  function play() {
    hideInfo();
    $('intro').classList.add('gone');
    if (phase === 'explore' || phase === 'toTop' || current < 0) flyTo(selected);
    setPlaying(true);
  }

  $('play').addEventListener('click', () => playing ? setPlaying(false) : play());
  $('mpause').addEventListener('click', () => playing ? setPlaying(false) : play());
  $('expand').addEventListener('click', () => root.classList.remove('min'));
  $('collapse').addEventListener('click', () => root.classList.add('min'));
  $('next').addEventListener('click', next);
  $('mnext').addEventListener('click', next);
  $('prev').addEventListener('click', prev);
  $('info').addEventListener('click', () => showInfo(current >= 0 ? current : selected));
  $('minfo').addEventListener('click', () => showInfo(current >= 0 ? current : selected));
  $('overview').addEventListener('click', () => { setPlaying(false); hideInfo(); toTop(); });
  chips.forEach((c, i) => c.addEventListener('click', () => { flyTo(i); showInfo(i); }));

  applyPose(TOP);
